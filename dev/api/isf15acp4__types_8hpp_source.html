431constexpr Error Validate(const ButtonConfig& b) noexcept {
432 if ((b.debounce_ms == 0U) || (b.debounce_ms > 1000U)) {
433 return Error::InvalidArgument;
434 }
435 if ((b.long_press_ms <= b.debounce_ms) || (b.long_press_ms > kButtonTimeMaxMs)) {
436 return Error::InvalidArgument;
437 }
438 if (b.repeat_interval_ms != 0U) {
439 if ((b.repeat_interval_ms < 10U) || (b.repeat_interval_ms > 10000U)) {
440 return Error::InvalidArgument;
441 }
442 }
443 if (b.repeat_delay_ms > kButtonTimeMaxMs) {
444 return Error::InvalidArgument;
445 }
446 return Error::Ok;
447}

452constexpr bool DisplayFieldsValid(const Config& config) noexcept {
453 const bool enums =
454 IsEnumerator(config.variant) && IsEnumerator(config.remap.depth) && IsEnumerator(config.mode);
455 const bool rows = (config.start_line < reg::kRamRows) && (config.display_offset < reg::kRamRows);
456 const bool dim_ok = (config.dim.contrast_a <= config.contrast_a) &&
457 (config.dim.contrast_b <= config.contrast_b) &&
458 (config.dim.contrast_c <= config.contrast_c);
459 const bool levels = dim_ok && (config.master_current <= reg::kMasterCurrentMax) &&
460 (config.precharge_level <= reg::kPrechargeSteps) &&
461 (config.vcomh_level <= reg::kPrechargeSteps) &&
462 (config.dim.precharge <= reg::kPrechargeSteps);
463 return enums && rows && levels;
464}
465
467constexpr bool ClockFieldsValid(const Config& config) noexcept {
468 const bool phases = (config.phase1 != 0U) && (config.phase1 <= reg::kPhaseMax) &&
469 (config.phase2 != 0U) && (config.phase2 <= reg::kPhaseMax);
470 return phases && (config.clock_divide <= 15U) && (config.clock_frequency <= 15U);
471}
472
474constexpr bool TimeKnobsValid(const Config& config) noexcept {
475 const bool waits = (config.vcc_settle_ms != 0U) && (config.vcc_settle_ms <= kWaitMaxMs) &&
476 (config.vcc_off_ms <= kWaitMaxMs) &&
477 (config.engine_settle_us <= kEngineSettleMaxUs);
478 const bool idle = (config.idle_dim_ms <= (kButtonTimeMaxMs * 100U)) &&
479 (config.idle_sleep_ms <= (kButtonTimeMaxMs * 100U));
480 const bool ordered = (config.idle_dim_ms == 0U) || (config.idle_sleep_ms == 0U) ||
481 (config.idle_sleep_ms > config.idle_dim_ms);
482 return waits && idle && ordered;
483}
