133constexpr PixelBytes EncodePixel(ColorDepth depth, Color color, bool blue_first) noexcept {
134 const Channels ch = Split(color);
135 const uint32_t first = blue_first ? ch.blue : ch.red; // colour C, 5 bits
136 const uint32_t second = ch.green; // colour B, 6 bits
137 const uint32_t third = blue_first ? ch.red : ch.blue; // colour A, 5 bits
138 PixelBytes p{};
139 switch (depth) {
140 case ColorDepth::Colors256:
141 p.bytes[0] =
142 static_cast<uint8_t>(((first >> 2U) << 5U) | ((second >> 3U) << 2U) | (third >> 3U));
143 p.count = 1U;
144 break;
145 case ColorDepth::Colors65k:
146 p.bytes[0] = static_cast<uint8_t>((first << 3U) | (second >> 3U));
147 p.bytes[1] = static_cast<uint8_t>(((second & 0x07U) << 5U) | third);
148 p.count = 2U;
149 break;
150 case ColorDepth::Colors65kFormat2:
151 p.bytes[0] = static_cast<uint8_t>(first << 1U);
152 p.bytes[1] = static_cast<uint8_t>(second);
153 p.bytes[2] = static_cast<uint8_t>(third << 1U);
154 p.count = 3U;
155 break;
156 default:
157 break;
158 }
159 return p;
160}
