82 constexpr ButtonUpdate Update(bool down, uint32_t now_ms) noexcept {
83 ButtonUpdate u{};
84 u.timestamp_ms = now_ms;
85 if (!started_) {
86 started_ = true;
87 pressed_ = down;
88 pressed_at_ms_ = now_ms;
89 changing_ = false;

91 }

93 if (!changing_) {

98 return accept(down, now_ms);
99 }
100 return u;
101 }
102 changing_ = false;

104 return holdEvent(now_ms);

106 return u;

119 constexpr ButtonUpdate accept(bool down, uint32_t now_ms) noexcept {
120 ButtonUpdate u{};
121 u.timestamp_ms = changed_at_ms_;
122 changing_ = false;
123 pressed_ = down;
124 if (down) {
125 pressed_at_ms_ = changed_at_ms_;
126 long_sent_ = false;
127 repeat_armed_ = (config_.repeat_interval_ms != 0U);
128 repeat_due_ms_ = config_.repeat_delay_ms;
129 press_count_ = press_count_ + 1U;
130 u.event = ButtonEvent::Pressed;
131 } else {
132 u.event = ButtonEvent::Released;
133 u.held_ms = changed_at_ms_ - pressed_at_ms_;
134 u.was_long = long_sent_;
135 repeat_armed_ = false;
136 }
137 static_cast<void>(now_ms);
138 return u;
139 }

141 constexpr ButtonUpdate holdEvent(uint32_t now_ms) noexcept {
142 ButtonUpdate u{};
143 u.timestamp_ms = now_ms;
144 const uint32_t held = now_ms - pressed_at_ms_;
145 u.held_ms = held;
146 if (!long_sent_ && (held >= config_.long_press_ms)) {
147 long_sent_ = true;
148 u.event = ButtonEvent::LongPress;
149 } else if (repeat_armed_ && (held >= repeat_due_ms_)) {
150 // The next repeat is one interval after this poll: a slow poll never bursts.
151 repeat_due_ms_ = held + config_.repeat_interval_ms;
152 u.event = ButtonEvent::Repeat;
153 } else {
154 // Nothing is due.
155 }
156 return u;
157 }
