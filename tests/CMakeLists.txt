# =============================================================================
# ISF15ACP4 driver: host tests
# =============================================================================
# The driver against a register model of the chip, with the firmware's flags
# (-fno-exceptions -fno-rtti) and warnings as errors. No hardware, no ESP-IDF:
#
#   cmake -S . -B build -DISF15ACP4_BUILD_TESTS=ON
#   cmake --build build
#   ctest --test-dir build --output-on-failure
# =============================================================================

# ISO C++20, never the GNU dialect (handbook KB-CPP20-001): set here too, so that this folder is strict
# whoever includes it.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# -pedantic-errors and the C++20 deprecations as errors (KB-CPP20-001, -003). -Wswitch-enum: a switch over
# an enumeration ends with `default: break;` (MISRA C++:2023 Rule 9.4.2), and a new enumerator still warns.
set(ISF15ACP4_TEST_FLAGS -fno-exceptions -fno-rtti -Wall -Wextra -Wpedantic -pedantic-errors -Wshadow
    -Wconversion -Wsign-conversion -Wdeprecated -Wdeprecated-enum-enum-conversion
    -Wdeprecated-enum-float-conversion -Wimplicit-fallthrough -Wswitch-enum -Werror)
# The standard library's own bounds checks, so an out-of-range span or array access traps in the tests
# and fuzzers (KB-CPP20-043).
set(ISF15ACP4_TEST_DEFINITIONS _GLIBCXX_ASSERTIONS=1 _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST)

function(isf15acp4_test_target name)
  target_link_libraries(${name} PRIVATE isf15acp4::isf15acp4)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_options(${name} PRIVATE ${ISF15ACP4_TEST_FLAGS})
  target_compile_definitions(${name} PRIVATE ${ISF15ACP4_TEST_DEFINITIONS})
endfunction()

# --- Strict-dialect header checks ---------------------------------------------
# One translation unit per public header (and through the driver header its src/ .ipp): each header
# compiles on its own, first in its unit, as ISO C++20 with -pedantic-errors, and the unit asserts
# that the dialect really is C++20 (handbook KB-CPP20-001, checks strict-dialect and
# header-self-contained). They build with the host tests on GCC and Clang and, as part of
# isf15acp4_optimised_build, in the Cortex-M cross-build.
file(GLOB_RECURSE ISF15ACP4_PUBLIC_HEADERS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/inc/*.hpp
  ${PROJECT_SOURCE_DIR}/inc/*.h)
set(ISF15ACP4_HEADER_CHECKS)
foreach(header ${ISF15ACP4_PUBLIC_HEADERS} ${ISF15ACP4_VERSION_HEADER})
  if(header STREQUAL ISF15ACP4_VERSION_HEADER)
    file(RELATIVE_PATH rel ${ISF15ACP4_VERSION_HEADER_DIR} ${header})
  else()
    file(RELATIVE_PATH rel ${PROJECT_SOURCE_DIR}/inc ${header})
  endif()
  string(MAKE_C_IDENTIFIER "${rel}" id)
  set(check ${CMAKE_CURRENT_BINARY_DIR}/header_checks/${id}.cpp)
  string(CONCAT content "// Generated by tests/CMakeLists.txt: ${rel} compiles on its own as ISO C++20.\n"
    "#include \"${rel}\"\n"
    "static_assert(__cplusplus == 202002L, \"the strict-dialect check must build as -std=c++20\");\n")
  if(EXISTS ${check})
    file(READ ${check} old)
  else()
    set(old "")
  endif()
  if(NOT old STREQUAL content)
    file(WRITE ${check} "${content}")
  endif()
  list(APPEND ISF15ACP4_HEADER_CHECKS ${check})
endforeach()
add_library(isf15acp4_header_checks OBJECT ${ISF15ACP4_HEADER_CHECKS})
isf15acp4_test_target(isf15acp4_header_checks)

# --- Version: the getters, and every copy of the version string ---------------
function(isf15acp4_version_test name include_dir major minor patch)
  add_executable(${name} test_version.cpp)
  if(include_dir)
    target_include_directories(${name} BEFORE PRIVATE ${include_dir})
  endif()
  isf15acp4_test_target(${name})
  target_compile_definitions(${name} PRIVATE EXPECTED_MAJOR=${major} EXPECTED_MINOR=${minor}
    EXPECTED_PATCH=${patch} EXPECTED_STRING="${major}.${minor}.${patch}")
  add_test(NAME ${name} COMMAND ${name})
endfunction()

isf15acp4_version_test(isf15acp4_version "" ${PROJECT_VERSION_MAJOR} ${PROJECT_VERSION_MINOR}
  ${PROJECT_VERSION_PATCH})

# The same template with another version, found ahead of the real header.
function(isf15acp4_version_probe_header)
  set(ISF15ACP4_VERSION_MAJOR 9)
  set(ISF15ACP4_VERSION_MINOR 8)
  set(ISF15ACP4_VERSION_PATCH 7)
  set(ISF15ACP4_VERSION 9.8.7)
  configure_file(${ISF15ACP4_VERSION_TEMPLATE} ${CMAKE_CURRENT_BINARY_DIR}/version_probe/isf15acp4_version.h @ONLY)
endfunction()
isf15acp4_version_probe_header()
isf15acp4_version_test(isf15acp4_version_probe ${CMAKE_CURRENT_BINARY_DIR}/version_probe 9 8 7)

# Files that carry their own copy of the version must match cmake/isf15acp4_build_settings.cmake.
foreach(copy "_config/Doxyfile|PROJECT_NUMBER")
  string(REPLACE "|" ";" parts "${copy}")
  list(GET parts 0 file)
  list(GET parts 1 key)
  string(MAKE_C_IDENTIFIER "${file}" id)
  add_test(NAME isf15acp4_version_copy_${id}
    COMMAND ${CMAKE_COMMAND} -DFILE=${PROJECT_SOURCE_DIR}/${file} -DKEY=${key} -DVERSION=${PROJECT_VERSION}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/check_version_copy.cmake)
endforeach()

# --- Compile-time contract: these must fail to build, for the reason in their EXPECT: line ---
file(GLOB ISF15ACP4_COMPILE_FAIL CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/compile_fail/*.cpp)
foreach(src ${ISF15ACP4_COMPILE_FAIL})
  get_filename_component(stem ${src} NAME_WE)
  file(STRINGS ${src} expect REGEX "EXPECT: ")
  string(REGEX REPLACE ".*EXPECT: " "" expect "${expect}")
  add_executable(isf15acp4_compile_fail_${stem} EXCLUDE_FROM_ALL ${src})
  isf15acp4_test_target(isf15acp4_compile_fail_${stem})
  add_test(NAME isf15acp4_compile_fail_${stem}
    COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target isf15acp4_compile_fail_${stem})
  # Pass only when the compiler rejects the code with the expected diagnostic, not another error.
  # Each one builds a target of this build tree: run them one at a time, so that a stale tree is not
  # regenerated by several of them at once (seen with ctest -j after a CMakeLists change).
  set_tests_properties(isf15acp4_compile_fail_${stem} PROPERTIES PASS_REGULAR_EXPRESSION "${expect}"
    RESOURCE_LOCK isf15acp4_build_tree)
endforeach()
